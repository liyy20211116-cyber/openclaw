# Unattended Health Guardian Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a watchdog so Jarvis One Company OS can detect failed unattended jobs, rerun safe read-only checks, write audit evidence, and produce a CEO-readable health report without waiting for manual supervision.

**Architecture:** Add a small health guardian around the existing Windows Scheduled Tasks and `scripts/xhs_post_monitor.py`. The guardian reads scheduler status, monitor event logs, CSV metrics, and SQLite task logs, then writes a health report and DB evidence. It may only rerun read-only monitoring jobs; it must never publish, comment, message, or book revenue.

**Tech Stack:** Python 3.13 with Playwright already installed, PowerShell Scheduled Tasks, SQLite, JSON/CSV files, existing Jarvis DB schema (`tasks`, `task_logs`, `workflow_runs`, `workflow_steps`, `agent_memories`).

```python
from __future__ import annotations

import csv
import datetime as dt
import json
import sqlite3
import subprocess
from pathlib import Path
from typing import Any

ROOT = Path(__file__).resolve().parents[1]
DB_PATHS = [
    ROOT / "jarvis-one-company-os" / "dev.db",
    Path.home() / "AppData" / "Roaming" / "jarvis-one-company-os" / "company-data" / "dev.db",
]


def classify_task_result(result: int) -> str:
    if result == 0:
        return "ok"
    if result == 267011:
        return "pending"
    if result == 2147942402:
        return "failed_path_or_python"
    return "failed"


def classify_event_freshness(last_seen: dt.datetime | None, now: dt.datetime, max_minutes: int) -> str:
    if last_seen is None:
        return "missing"
    age = now - last_seen
    return "fresh" if age <= dt.timedelta(minutes=max_minutes) else "stale"


def render_report(now: dt.datetime, checks: list[dict[str, Any]], events: list[dict[str, Any]], actions: list[str]) -> str:
    lines = [
        "# 无人值守健康守护报告",
        "",
        f"生成时间：{now.strftime('%Y-%m-%d %H:%M:%S')}",
        "",
        "## 调度任务",
    ]
    for check in checks:
        lines.append(f"- {check['name']}：state={check['state']}，result={check['result']}")
    lines.extend(["", "## 最近事件"])
    if not events:
        lines.append("- 未发现监控事件")
    for event in events:
        lines.append(
            f"- {event.get('event', '')}：阅读 {event.get('views', 0)}，点赞 {event.get('likes', 0)}，"
            f"收藏 {event.get('favorites', 0)}，评论 {event.get('comments', 0)}"
        )
    lines.extend(["", "## 自动动作"])
    if not actions:
        lines.append("- 无自动补跑")
    for action in actions:
        lines.append(f"- {action}")
    lines.extend(
        [
            "",
            "## 边界",
            "- 只允许只读监控和补跑指标读取。",
            "- 不允许自动发布、自动评论、自动私信。",
            "- 未到账不得写入收入。",
        ]
    )
    return "\n".join(lines) + "\n"
```

```python
def read_recent_events(path: Path, limit: int = 5) -> list[dict[str, Any]]:
    if not path.exists():
        return []
    rows: list[dict[str, Any]] = []
    for line in path.read_text(encoding="utf-8").splitlines():
        if not line.strip():
            continue
        rows.append(json.loads(line))
    return rows[-limit:]


def get_scheduled_task_info(task_name: str) -> dict[str, Any]:
    command = [
        "powershell",
        "-NoProfile",
        "-Command",
        (
            f"$i=Get-ScheduledTaskInfo -TaskName '{task_name}'; "
            f"$t=Get-ScheduledTask -TaskName '{task_name}'; "
            "[pscustomobject]@{"
            "TaskName='" + task_name + "';"
            "State=$t.State.ToString();"
            "LastRunTime=$i.LastRunTime.ToString('s');"
            "NextRunTime=$i.NextRunTime.ToString('s');"
            "LastTaskResult=$i.LastTaskResult"
            "} | ConvertTo-Json -Compress"
        ),
    ]
    completed = subprocess.run(command, cwd=ROOT, capture_output=True, text=True, check=False)
    if completed.returncode != 0:
        return {"name": task_name, "state": "Missing", "result": "failed", "raw": completed.stderr.strip()}
    payload = json.loads(completed.stdout)
    return {
        "name": task_name,
        "state": payload.get("State", ""),
        "last_run_time": payload.get("LastRunTime", ""),
        "next_run_time": payload.get("NextRunTime", ""),
        "result_code": int(payload.get("LastTaskResult", 1)),
        "result": classify_task_result(int(payload.get("LastTaskResult", 1))),
    }
```

```python
def write_db_log(db_path: Path, detail: dict[str, Any]) -> None:
    if not db_path.exists():
        return
    con = sqlite3.connect(db_path)
    try:
        con.execute(
            "insert or replace into task_logs (id, taskId, operatorId, actionType, detailJson, createdAt) values (?,?,?,?,?,?)",
            (
                "task_log_unattended_guardian_latest",
                "task_xhs_salary_post2_ops_20260429",
                "agent_jarvis",
                "complete",
                json.dumps(detail, ensure_ascii=False),
                dt.datetime.now().isoformat(timespec="seconds"),
            ),
        )
        con.commit()
    finally:
        con.close()


def run_catchup(python_executable: str, command_text: str) -> dict[str, Any]:
    if "xhs_post_monitor.py" not in command_text:
        return {"ok": False, "reason": "catchup command is not an approved read-only monitor"}
    if any(blocked in command_text for blocked in ["publish", "comment_reply", "private_message", "revenue"]):
        return {"ok": False, "reason": "catchup command contains blocked action"}
    parts = [python_executable] + command_text.split()
    completed = subprocess.run(parts, cwd=ROOT, capture_output=True, text=True, check=False)
    return {
        "ok": completed.returncode == 0,
        "returncode": completed.returncode,
        "stdout": completed.stdout[-2000:],
        "stderr": completed.stderr[-2000:],
    }
```

```python
def load_config() -> dict[str, Any]:
    return json.loads((ROOT / "config" / "unattended-health-guardian.json").read_text(encoding="utf-8"))


def main() -> int:
    config = load_config()
    now = dt.datetime.now()
    checks: list[dict[str, Any]] = []
    actions: list[str] = []
    events: list[dict[str, Any]] = []

    for job in config["jobs"]:
        for task_name in job["scheduled_tasks"]:
            checks.append(get_scheduled_task_info(task_name))
        event_path = ROOT / job["events_file"]
        events.extend(read_recent_events(event_path, limit=5))

    health = {
        "generated_at": now.isoformat(timespec="seconds"),
        "checks": checks,
        "events": events,
        "actions": actions,
        "mode": config["mode"],
    }
    report = render_report(now=now, checks=checks, events=events, actions=actions)
    latest_path = ROOT / config["report"]["latest_path"]
    latest_path.parent.mkdir(parents=True, exist_ok=True)
    latest_path.write_text(report, encoding="utf-8")
    history_dir = ROOT / config["report"]["history_dir"]
    history_dir.mkdir(parents=True, exist_ok=True)
    (history_dir / f"unattended-health-{now.strftime('%Y%m%d-%H%M%S')}.md").write_text(report, encoding="utf-8")

    for db_path in DB_PATHS:
        write_db_log(db_path, health)

    print(json.dumps({"ok": True, "report": str(latest_path), "checks": len(checks), "events": len(events)}, ensure_ascii=False))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
